\block{software}
% -------------------------------------------------------

\begin{itemize}

\item \href{https://pypi.org/project/pybdiff/}{pybdiff}\\
  Python replication of Stata's \texttt{bdiff} command. Tests whether regression
  coefficients differ across two subgroups via permutation, bootstrap, or Wald tests,
  with optional parallel execution. Install via \texttt{pip install pybdiff}.

\item \href{https://pypi.org/project/pyshapley2/}{pyshapley2}\\
  Python replication of Stata's \texttt{shapley2} command. Computes the Shapley--Owen
  decomposition of any regression fit statistic across independent variables or
  user-defined groups. Install via \texttt{pip install pyshapley2}.

\item \href{https://pypi.org/project/pybacondecomp/}{pybacondecomp}\\
  Python replication of Stata's \texttt{bacondecomp} command.\\
  Implements the Goodman--Bacon (2021) decomposition of TWFE DiD estimators
  into all 2$\times$2 comparisons.\\ Install via \texttt{pip install pybacondecomp}.

\item \href{https://jrnldaily.com}{jrnldaily.com}\\
  An automated economics literature platform that aggregates new working papers and
  journal articles daily, with AI-generated summaries and cross-language search.

\end{itemize}
